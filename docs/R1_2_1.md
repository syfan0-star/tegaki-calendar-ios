```swift
import StoreKit
import SwiftUI

/// 1.2.1 (docs/R1_2_1.md §2): asks StoreKit for the rating prompt each time `trigger` grows.
struct ReviewRequester: ViewModifier {
    let trigger: Int
    @Environment(\.requestReview) private var requestReview

    func body(content: Content) -> some View {
        content.onChange(of: trigger) {
            requestReview()
        }
    }
}
```

### 2.4 いつ・どこで（`CalendarRootScreen.swift` の今の行）

```swift
private func move(by delta: Int) {
    let wroteHere = inkChangedThisVisit
    let lastInk = lastInkChangeAt
    show(kind: kind, focus: PageNavigation.move(kind, focus: focus, by: delta))
    scheduleReviewRequest(leftPageWithInk: wroteHere, lastInkChangeAt: lastInk)
}

/// 1.2.1 (docs/R1_2_1.md §2): after ◀ ▶ from a page written on, a pause on the new page, then the
/// rating prompt when everything in §2.3 holds. Nothing in screenshot mode, the XCTest host or a sandbox build.
private func scheduleReviewRequest(leftPageWithInk: Bool, lastInkChangeAt: Date?) {
    let now = Date()
    guard Self.reviewAllowedHere(screenshot: ScreenshotMode.isActive, testHost: SyncEngine.isTestHost, sandbox: InkTrace.isAvailable),
          ReviewPrompt.isCalmMoment(leftPageWithInk: leftPageWithInk,
                                    secondsSinceInk: lastInkChangeAt.map { now.timeIntervalSince($0) },
                                    secondsActive: now.timeIntervalSince(activeSince)),
          ReviewPrompt.isDue(AppSettings.reviewRecord, today: DayKey(now), version: Self.marketingVersion, access: access)
    else { return }
    reviewTask?.cancel()
    reviewTask = Task { @MainActor in
        try? await Task.sleep(for: .seconds(ReviewPrompt.pauseOnNewPage))
        let today = DayKey(Date())
        guard !Task.isCancelled,
              Self.reviewCanShowNow(active: UIApplication.shared.applicationState == .active, sheetIsUp: sheetIsUp,
                                    accountQuestion: showAccountChange, saveFailed: saveError != nil,
                                    onboardingDone: onboardingDone, toastShowing: toast != nil),
              ReviewPrompt.isDue(AppSettings.reviewRecord, today: today, version: Self.marketingVersion, access: access)
        else { return }
        // Recorded before the call: StoreKit never says whether it showed the prompt (Apple's sample does the same).
        AppSettings.reviewRecord = ReviewPrompt.requested(AppSettings.reviewRecord, today: today, version: Self.marketingVersion)
        reviewRequestTrigger += 1
    }
}

static func reviewAllowedHere(screenshot: Bool, testHost: Bool, sandbox: Bool) -> Bool {
    !screenshot && !testHost && !sandbox
}

static func reviewCanShowNow(active: Bool, sheetIsUp: Bool, accountQuestion: Bool, saveFailed: Bool,
                             onboardingDone: Bool, toastShowing: Bool) -> Bool {
    active && !sheetIsUp && !accountQuestion && !saveFailed && onboardingDone && !toastShowing
}

static var marketingVersion: String {
    (Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String) ?? ""
}
```

`InkTrace.isAvailable` と `SyncEngine.isTestHost` は読むだけです（どちらも凍結の 18 ファイルの中なので変えない）。

```swift
public struct ReviewRecord: Equatable, Sendable {
    public var firstUseDay: DayKey?          // 初めて使った日（1.2 からの人は 1.2.1 の最初の起動の日）
    public var inkDays: Int                  // 自分で書いた日の数（前のお願いの後から）
    public var lastInkDay: DayKey?           // 最後に数えた日
    public var lastRequestVersion: String?   // 前にお願いした版
    public var lastRequestDay: DayKey?       // 前にお願いした日
    public init(firstUseDay: DayKey? = nil, inkDays: Int = 0, lastInkDay: DayKey? = nil,
                lastRequestVersion: String? = nil, lastRequestDay: DayKey? = nil)
}

public enum ReviewPrompt {
    public static let inkDaysNeeded = 3
    public static let daysSinceFirstUse = 7
    public static let daysBetweenRequests = 90
    public static let quietAfterInk: TimeInterval = 3
    public static let activeAtLeast: TimeInterval = 60
    public static let pauseOnNewPage: Double = 2

    /// firstUseDay が nil なら today にする。あればそのまま
    public static func usedOn(_ record: ReviewRecord, today: DayKey) -> ReviewRecord
    /// usedOn の後、lastInkDay != today なら inkDays + 1・lastInkDay = today
    public static func inkWritten(_ record: ReviewRecord, today: DayKey) -> ReviewRecord
    /// access.canEdit、version が空でない、lastRequestVersion != version、inkDays >= 3、
    /// firstUseDay があり today.dayNumber − firstUseDay.dayNumber >= 7、
    /// lastRequestDay が nil か today.dayNumber − lastRequestDay.dayNumber >= 90
    public static func isDue(_ record: ReviewRecord, today: DayKey, version: String, access: AccessState) -> Bool
    /// usedOn の後、lastRequestVersion = version、lastRequestDay = today、inkDays = 0、lastInkDay = today
    /// （お願いした日に書いても数えない。次の版は新しく 3 日書いてから）
    public static func requested(_ record: ReviewRecord, today: DayKey, version: String) -> ReviewRecord
    /// leftPageWithInk、secondsSinceInk が nil でなく 3 以上、secondsActive が 60 以上
    public static func isCalmMoment(leftPageWithInk: Bool, secondsSinceInk: TimeInterval?, secondsActive: TimeInterval) -> Bool
}
```
